import React, { useState, useEffect } from "react";
import Header from "./components/Header";
import TaglineIntro from "./components/TaglineIntro";
import Hero from "./components/Hero";
import Myself from "./components/Myself";
import Experience from "./components/Experience";
import MyWork from "./components/MyWork";
import Reviews from "./components/Reviews";
import MyYouTube from "./components/MyYouTube";
import Contact from "./components/Contact";

function App() {
  const [activeSection, setActiveSection] = useState("home");
  const [introState, setIntroState] = useState<
    "visible" | "exiting" | "hidden"
  >("visible");

  useEffect(() => {
    // Prevent scrolling while the intro is visible or animating
    if (introState !== "hidden") {
      document.body.style.overflow = "hidden";
    } else {
      document.body.style.overflow = "";
    }
  }, [introState]);

  useEffect(() => {
    const handleScroll = () => {
      const sections = [
        "home",
        "myself",
        "experience",
        "work",
        "reviews",
        "youtube",
        "contact",
      ];
      const scrollPosition = window.scrollY + 100;

      for (const section of sections) {
        const element = document.getElementById(section);
        if (element) {
          const { offsetTop, offsetHeight } = element;
          if (
            scrollPosition >= offsetTop &&
            scrollPosition < offsetTop + offsetHeight
          ) {
            setActiveSection(section);
            break;
          }
        }
      }
    };

    window.addEventListener("scroll", handleScroll);
    return () => {
      window.removeEventListener("scroll", handleScroll);
    };
  }, []);

  const handleAnimationComplete = () => {
    setIntroState("exiting"); // 1. Trigger the exit animations
    setTimeout(() => {
      setIntroState("hidden"); // 2. After animation, remove the intro component
    }, 1200); // This duration MUST match the CSS animation duration
  };

  const renderIntro = () => {
    if (introState === "hidden") {
      return null;
    }

    // CRITICAL FIX: The wrapper is now a fixed, full-screen overlay.
    // This prevents the main app content from pushing it up.
    return (
      <div
        className={`fixed inset-0 z-50 ${
          introState === "exiting" ? "animate-intro-zoom-out" : ""
        }`}
      >
        <TaglineIntro onAnimationComplete={handleAnimationComplete} />
      </div>
    );
  };

  return (
    <div className="min-h-screen bg-background">
      {renderIntro()}

      {/* 
        This div is initially invisible. When the intro starts exiting, 
        it becomes visible and plays its zoom-in animation.
      */}
      <div
        className={
          introState !== "visible" ? "animate-hero-zoom-in" : "opacity-0"
        }
      >
        <Header activeSection={activeSection} />

        <main className="relative">
          <section id="home">
            <Hero />
          </section>

          <section id="myself">
            <Myself />
          </section>

          <section id="experience">
            <Experience />
          </section>

          <section id="work">
            <MyWork />
          </section>

          <section id="reviews">
            <Reviews />
          </section>

          <section id="youtube">
            <MyYouTube />
          </section>

          <section id="contact">
            <Contact />
          </section>
        </main>
      </div>
    </div>
  );
}

export default App;

import React from "react";
import { Home, User, Briefcase, Play, Star, Youtube, Mail } from "lucide-react";
import { useState, useEffect, useRef } from "react";

interface NavigationItem {
  id: string;
  label: string;
  icon: React.ReactNode;
  color: string;
}

interface CircularNavigationProps {
  isOpen: boolean;
  onClose: () => void;
  activeSection: string;
}

// --- CONFIGURATION ---
// You can change this value to make the rotation faster or slower.
// Positive values rotate clockwise, negative values counter-clockwise.
const rotationSpeed = 8; // Degrees per second
// --- END CONFIGURATION ---

const navigationItems: NavigationItem[] = [
  {
    id: "home",
    label: "Home",
    icon: <Home size={20} />,
    color: "from-accent to-primary",
  },
  {
    id: "myself",
    label: "Myself",
    icon: <User size={20} />,
    color: "from-primary to-accent",
  },
  {
    id: "experience",
    label: "Experience",
    icon: <Briefcase size={20} />,
    color: "from-accent to-secondary",
  },
  {
    id: "work",
    label: "My Work",
    icon: <Play size={20} />,
    color: "from-primary to-accent-dark",
  },
  {
    id: "reviews",
    label: "Reviews",
    icon: <Star size={20} />,
    color: "from-accent-dark to-primary",
  },
  {
    id: "youtube",
    label: "My YouTube",
    icon: <Youtube size={20} />,
    color: "from-accent to-accent-dark",
  },
  {
    id: "contact",
    label: "Contact",
    icon: <Mail size={20} />,
    color: "from-primary to-accent",
  },
];

const CircularNavigation: React.FC<CircularNavigationProps> = ({
  isOpen,
  onClose,
  activeSection,
}) => {
  const [rotation, setRotation] = useState(0);
  const [hoveredId, setHoveredId] = useState<string | null>(null);
  const animationFrameRef = useRef<number>();

  useEffect(() => {
    // Animation loop for rotation
    const animate = () => {
      setRotation((prevRotation) => (prevRotation + rotationSpeed / 60) % 360);
      animationFrameRef.current = requestAnimationFrame(animate);
    };

    if (isOpen) {
      // Start the animation when the menu opens
      animationFrameRef.current = requestAnimationFrame(animate);
    }

    // Cleanup function to stop the animation
    return () => {
      if (animationFrameRef.current) {
        cancelAnimationFrame(animationFrameRef.current);
      }
    };
  }, [isOpen]);

  const scrollToSection = (sectionId: string) => {
    const element = document.getElementById(sectionId);
    if (element) {
      element.scrollIntoView({ behavior: "smooth" });
      onClose();
    }
  };

  const radius = 200;
  const centerX = 300;
  const centerY = 300;

  return (
    <div
      className={`fixed inset-0 z-40 transition-all duration-500 ${
        isOpen ? "opacity-100 visible" : "opacity-0 invisible"
      }`}
    >
      {/* Background overlay */}
      <div
        className={`absolute inset-0 bg-background/50 backdrop-blur-sm transition-opacity duration-500 ${
          isOpen ? "opacity-100" : "opacity-0"
        }`}
        onClick={onClose}
      ></div>

      {/* Circular menu */}
      <div className="absolute top-0 right-0 w-[600px] h-[600px] overflow-hidden">
        <div
          className="relative w-full h-full"
          style={{ transform: `rotate(${rotation}deg)` }} // Apply rotation to the main container
        >
          {navigationItems.map((item, index) => {
            const angle = (index * 360) / navigationItems.length - 90;
            const radian = (angle * Math.PI) / 180;
            const x = centerX + radius * Math.cos(radian);
            const y = centerY + radius * Math.sin(radian);

            // Determine dynamic transform values
            const isHovered = hoveredId === item.id;
            const buttonScale = isOpen ? (isHovered ? 1.25 : 1) : 0;
            const contentScale = isHovered ? 1.1 : 1;

            // CORRECTED: The content's transform combines counter-rotation and hover scale
            const contentTransform = `rotate(${-rotation}deg) scale(${contentScale})`;

            return (
              <button
                key={item.id}
                onClick={() => scrollToSection(item.id)}
                onMouseEnter={() => setHoveredId(item.id)}
                onMouseLeave={() => setHoveredId(null)}
                className={`absolute w-20 h-20 rounded-full bg-background/80 backdrop-blur-md border border-ui-border flex flex-col items-center justify-center text-primary text-xs font-medium transition-all duration-300 hover:bg-accent/20 hover:border-accent-light hover:shadow-accent group ${
                  isOpen ? "animate-fade-in" : ""
                }`}
                style={{
                  left: x - 40,
                  top: y - 40,
                  transform: `scale(${buttonScale})`, // The button itself only scales
                  transitionDelay: `${index * 50}ms`,
                }}
              >
                <div
                  className={`absolute inset-0 rounded-full bg-gradient-to-r ${item.color} opacity-0 group-hover:opacity-30 transition-all duration-300 group-hover:animate-pulse`}
                ></div>
                <div
                  className="relative z-10 flex flex-col items-center transition-transform duration-300"
                  style={{ transform: contentTransform }} // Apply counter-rotation and scale here
                >
                  {item.icon}
                  <span className="mt-1 text-[10px] group-hover:text-accent-light transition-colors duration-300">
                    {item.label}
                  </span>
                </div>
                {activeSection === item.id && (
                  <div
                    className={`absolute inset-0 rounded-full bg-gradient-to-r ${item.color} opacity-40 animate-pulse`}
                  ></div>
                )}
              </button>
            );
          })}
        </div>
      </div>
    </div>
  );
};

export default CircularNavigation;

// A small component to handle the number counting animation
const NumberCounter: React.FC<{ target: number; isVisible: boolean }> = ({
  target,
  isVisible,
}) => {
  const [count, setCount] = useState(0);

  useEffect(() => {
    if (!isVisible) return;

    let start = 0;
    const duration = 2000; // 2 seconds
    const frameRate = 1000 / 60; // 60fps
    const totalFrames = Math.round(duration / frameRate);
    const increment = target / totalFrames;

    const counter = setInterval(() => {
      start += increment;
      if (start >= target) {
        setCount(target);
        clearInterval(counter);
      } else {
        setCount(Math.ceil(start));
      }
    }, frameRate);

    return () => clearInterval(counter);
  }, [isVisible, target]);

  return (
    <span className="text-2xl font-bold text-primary group-hover:text-accent-light transition-colors duration-300">
      {count}%
    </span>
  );
};

  const carouselRef = useRef<HTMLDivElement>(null); // Ref for the carousel container

  // Intersection Observer for initial fade-in animation
  useEffect(() => {
    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          setIsVisible(true);
          observer.disconnect();
        }
      },
      { threshold: 0.1 }
    );
    if (sectionRef.current) observer.observe(sectionRef.current);
    return () => {
      if (sectionRef.current) observer.unobserve(sectionRef.current);
    };
  }, []);

  // Scroll listener for 3D perspective effect
  useEffect(() => {
    const handleScroll = () => {
      if (!carouselRef.current) return;

      const rect = carouselRef.current.getBoundingClientRect();
      const viewportHeight = window.innerHeight;
      const elementCenterY = rect.top + rect.height / 2;

      // Calculate position relative to viewport center (-1 to 1)
      const normalizedPosition =
        (elementCenterY - viewportHeight / 2) / (viewportHeight / 2);

      const maxRotation = 10; // Max tilt in degrees
      const newRotation = Math.max(
        -maxRotation,
        Math.min(maxRotation, normalizedPosition * maxRotation)
      );

      setScrollRotation(newRotation);
    };

    window.addEventListener("scroll", handleScroll, { passive: true });
    return () => window.removeEventListener("scroll", handleScroll);

// `src/components/TaglineIntro.tsx`
import React, { useEffect, useState } from "react";

interface TaglineIntroProps {
  onAnimationComplete: () => void;
}

const styledWords = [
  { text: "Every ", className: "text-accent" },
  { text: "frame, ", className: "text-primary" },
  { text: "can ", className: "text-secondary" },
  { text: "make ", className: "text-primary" },
  { text: "a ", className: "text-secondary" },
  { text: "story", className: "text-accent" },
];

const fullTagline = styledWords.map((w) => w.text).join("");
const typingSpeed = 60; // milliseconds per character

const TaglineIntro: React.FC<TaglineIntroProps> = ({ onAnimationComplete }) => {
  const [typedText, setTypedText] = useState("");
  const [isComplete, setIsComplete] = useState(false);

  useEffect(() => {
    // Typewriter effect
    const typingInterval = setInterval(() => {
      setTypedText((prev) => {
        if (prev.length < fullTagline.length) {
          return fullTagline.substring(0, prev.length + 1);
        } else {
          clearInterval(typingInterval);
          setIsComplete(true);
          return prev;
        }
      });
    }, typingSpeed);

    return () => clearInterval(typingInterval);
  }, []);

  useEffect(() => {
    // Notify parent component on completion after a delay
    if (isComplete) {
      const delayBeforeNotify = 500; // Reduced delay to make transition snappier

      const animationCompleteTimer = setTimeout(() => {
        onAnimationComplete();
      }, delayBeforeNotify);

      return () => {
        clearTimeout(animationCompleteTimer);
      };
    }
  }, [isComplete, onAnimationComplete]);

  // Renders the text with appropriate colors as it's being typed
  const renderTypedText = () => {
    const elements: React.ReactNode[] = [];
    let textIndex = 0;

    for (const { text: wordText, className } of styledWords) {
      const partLength = Math.min(
        wordText.length,
        Math.max(0, typedText.length - textIndex)
      );

      if (partLength > 0) {
        elements.push(
          <span key={textIndex} className={className}>
            {wordText.substring(0, partLength)}
          </span>
        );
      }

      textIndex += wordText.length;
      if (typedText.length < textIndex) break;
    }
    return elements;
  };

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center bg-background">
      <div className="text-center">
        <h1 className="text-6xl md:text-7xl font-bold leading-tight font-mono">
          {renderTypedText()}
          {!isComplete && (
            <span className="cursor-blink text-accent glow-red ml-1 text-7xl md:text-9xl relative -top-1">
              |
            </span>
          )}
        </h1>

        {/* Animated line */}
        <div className="mt-8 flex justify-center">
          <div
            className={`h-1 bg-gradient-to-r from-accent to-primary transition-all ease-in-out duration-1000 ${
              isComplete ? "w-64" : "w-0"
            }`}
          />
        </div>
      </div>

      {/* Particles effect */}
      <div className="absolute inset-0 overflow-hidden pointer-events-none">
        {Array.from({ length: 20 }).map((_, i) => (
          <div
            key={i}
            className="absolute w-1 h-1 bg-accent rounded-full animate-pulse"
            style={{
              left: `${Math.random() * 100}%`,
              top: `${Math.random() * 100}%`,
              animationDelay: `${Math.random() * 3}s`,
              animationDuration: `${2 + Math.random() * 2}s`,
            }}
          />
        ))}
      </div>
    </div>
  );
};

export default TaglineIntro;
